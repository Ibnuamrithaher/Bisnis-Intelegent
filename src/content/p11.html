🧾 Measure operasional

Jumlah Pengiriman = COUNTROWS ( Logistik )
Terkirim = CALCULATE ( [Jumlah Pengiriman], Logistik[Status] = "Terkirim" )
Terkirim Tepat Waktu = CALCULATE ( [Jumlah Pengiriman], Logistik[Status] = "Terkirim",
                         FILTER ( Logistik, Logistik[AktualHari] <= Logistik[EstimasiHari] ) )
On-time % = DIVIDE ( [Terkirim Tepat Waktu], [Terkirim] )
Error Rate = DIVIDE ( CALCULATE ( [Jumlah Pengiriman], Logistik[Status] = "Gagal/Retur" ), [Jumlah Pengiriman] )
Rata-rata Hari Kirim = AVERAGE ( Logistik[AktualHari] )
Status Stok = IF ( MIN ( Stok[Stok] ) < MIN ( Stok[StokMinimum] ), "Kritis", "Aman" )
