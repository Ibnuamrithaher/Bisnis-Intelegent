🧾 Measure yang perlu dibuat

Biaya Iklan = SUM ( Marketing[Biaya] )
Pendapatan Iklan = SUM ( Marketing[Pendapatan] )
CTR = DIVIDE ( SUM ( Marketing[Klik] ), SUM ( Marketing[Impresi] ) )
CVR = DIVIDE ( SUM ( Marketing[Order] ), SUM ( Marketing[Klik] ) )
CAC = DIVIDE ( [Biaya Iklan], SUM ( Marketing[Order] ) )
ROAS = DIVIDE ( [Pendapatan Iklan], [Biaya Iklan] )
ROI = DIVIDE ( [Pendapatan Iklan] * 0.4 - [Biaya Iklan], [Biaya Iklan] )   -- margin kotor 40%
